37/*
38 * Sleep for 'secs' seconds.
39 */
40static inline
41void d_sleep(const uint secs) noexcept
42{
43 DAF_SAFE_ASSERT_RETURN(secs > 0,);
44
45 try {
46#ifdef DAF_OS_WINDOWS
47 ::Sleep(secs * 1000);
48#else
49 ::sleep(secs);
50#endif
51 } DAF_SAFE_EXCEPTION("d_sleep");
52}
53
54/*
55 * Sleep for 'msecs' milliseconds.
56 */
57static inline
58void d_msleep(const uint msecs) noexcept
59{
60 DAF_SAFE_ASSERT_RETURN(msecs > 0,);
61
62 try {
63#ifdef DAF_OS_WINDOWS
64 ::Sleep(msecs);
65#else
66 ::usleep(msecs * 1000);
67#endif
68 } DAF_SAFE_EXCEPTION("d_msleep");
69}
