30/**
31 OpenGL3 Graphics context.
32
33 This provides access to the program, shaders and uniforms used by the underlying DAF implementation.
34 */
35struct OpenGL3GraphicsContext : GraphicsContext
36{
37 /**
38 The OpenGL3 program used for this context.
39 It is activated automatically before any widget onDisplay() is called.
40 If changing the current OpenGL program make sure to revert back to this one at the end of your pipeline.
41
42 @code
43 // use custom program
44 glUseProgram(context.program);
45
46 // custom stuff here
47
48 // revert back
49 glUseProgram(context.program);
50 @endcode
51 */
52 GLuint program;
53
54 /**
55 A vec4 uniform used to set the next drawing color.
56
57 @code
58 const GLfloat color[4] = { red, green, blue, alpha };
59 glUniform4fv(context.color, 1, color);
60 @endcode
61 */
62 GLuint color;
63
64 /**
65 A vertex shader attribute directly linked to gl_Position.
66 Use this to set the bounds for drawing, normalized as -1.0 to +1.0.
67 The @a width and @a height provide the total window size for convenience.
68
69 @code
70 const GLfloat triangle[] = { x1, y1, x2, y2, x3, y3 };
71 glEnableVertexAttribArray(context.bounds);
72 glVertexAttribPointer(context.bounds, 2, GL_FLOAT, GL_FALSE, 0, triangle);
73 @endcode
74 */
75 GLuint bounds;
76
77 /**
78 A vertex shader attribute directly linked to GL_TEXTURE0 map.
79 // TODO find the correct wording, map??.
80
81 @code
82 const GLfloat map[] = { 0.f, 0.f, 0.f, 1.f, 1.f, 1.f, 1.f, 0.f };
83 glEnableVertexAttribArray(context.textureMap);
84 glVertexAttribPointer(context.textureMap, 2, GL_FLOAT, GL_FALSE, 0, map);
85 @endcode
86 */
87 GLuint textureMap;
88
89 /**
90 A boolean uniform used to indicate if next drawing should @a texture or @a color.
91 Set to 0 for color mode, 1 for texture.
92 Default mode is color, if changed make sure to revert to color mode at the end of your pipeline.
93
94 @code
95 // setup for drawing based on texture
96 glUniform1i(context.usingTexture, 1);
97
98 // bind texture
99 glBindTexture(GL_TEXTURE_2D, myTextureId);
100 // etc..
101
102 // glDrawElements or similar
103
104 // unbind texture
105 glBindTexture(GL_TEXTURE_2D, 0);
106 // etc..
107
108 // revert to color mode
109 glUniform1i(context.usingTexture, 0);
110 @endcode
111 */
112 GLuint usingTexture;
113
114 /**
115 Set of buffers created with glGenBuffers.
116 Used internally in DAF to draw generic shapes, can be reused in custom code.
117 Unbound by default, make sure to leave them unbound at the end of your pipeline.
118 */
119 GLuint buffers[2];
120
121 /**
122 Vertex array object created with glGenVertexArrays, bound for the duration of onDisplay().
123 A core profile has no default vertex array object, so every glVertexAttribPointer and draw call needs a
124 real one bound or it fails with GL_INVALID_OPERATION and the draw is silently dropped. DAF binds this one
125 before calling onDisplay() and rebinds it around its own drawing, because NanoVG binds its own vertex
126 array object and leaves zero bound when it is done.
127 If changing the current vertex array object make sure to revert back to this one at the end of your
128 pipeline.
129 Always 0 when building for GLESv2, which has no vertex array objects.
130
131 @code
132 // use custom vertex array object
133 glBindVertexArray(myVertexArray);
134
135 // custom stuff here
136
137 // revert back
138 glBindVertexArray(context.vao);
139 @endcode
140 */
141 GLuint vao;
142
143 /**
144 Total width of the window used for this context.
145 */
146 uint width;
147
148 /**
149 Total height of the window used for this context.
150 */
151 uint height;
152};
153#endif

157static inline
158ImageFormat asDISTRHOImageFormat(const GLenum format)
159{
160 switch (format)
161 {
162 #if defined(DGL_USE_OPENGL3) && !defined(DGL_USE_GLES2)
163 case GL_RED:
164 #else
165 case GL_LUMINANCE:
166 #endif
167 return kImageFormatGrayscale;
168 #ifndef DGL_USE_GLES
169 case GL_BGR:
170 return kImageFormatBGR;
171 case GL_BGRA:
172 return kImageFormatBGRA;
173 #endif
174 case GL_RGB:
175 return kImageFormatRGB;
176 case GL_RGBA:
177 return kImageFormatRGBA;
178 }
179
180 return kImageFormatNull;
181}
182
183static inline
184GLenum asOpenGLImageFormat(const ImageFormat format)
185{
186 switch (format)
187 {
188 case kImageFormatNull:
189 break;
190 case kImageFormatGrayscale:
191 #if defined(DGL_USE_OPENGL3) && !defined(DGL_USE_GLES2)
192 return GL_RED;
193 #else
194 return GL_LUMINANCE;
195 #endif
196 case kImageFormatBGR:
197 #ifndef DGL_USE_GLES
198 return GL_BGR;
199 #else
200 return 0;
201 #endif
202 case kImageFormatBGRA:
203 #ifndef DGL_USE_GLES
204 return GL_BGRA;
205 #else
206 return 0;
207 #endif
208 case kImageFormatRGB:
209 return GL_RGB;
210 case kImageFormatRGBA:
211 return GL_RGBA;
212 }
213
214 return 0;
215}
