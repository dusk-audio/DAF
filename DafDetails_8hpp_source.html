345/**
346 Parameter ranges.@n
347 This is used to set the default, minimum and maximum values of a parameter.
348
349 By default a parameter has 0.0 as minimum, 1.0 as maximum and 0.0 as default.@n
350 When changing this struct values you must ensure maximum > minimum and default is within range.
351 */
352struct ParameterRanges {
353 /**
354 Default value.
355 */
356 float def;
357
358 /**
359 Minimum value.
360 */
361 float min;
362
363 /**
364 Maximum value.
365 */
366 float max;
367
368 /**
369 Default constructor, using 0.0 as default, 0.0 as minimum, 1.0 as maximum.
370 */
371 constexpr ParameterRanges() noexcept
372 : def(0.0f),
373 min(0.0f),
374 max(1.0f) {}
375
376 /**
377 Constructor using custom values.
378 */
379 constexpr ParameterRanges(const float df, const float mn, const float mx) noexcept
380 : def(df),
381 min(mn),
382 max(mx) {}
383
384 /**
385 Fix the default value within range.
386 */
387 void fixDefault() noexcept
388 {
389 fixValue(def);
390 }
391
392 /**
393 Fix a value within range.
394 */
395 void fixValue(float& value) const noexcept
396 {
397 if (value < min)
398 value = min;
399 else if (value > max)
400 value = max;
401 }
402
403 /**
404 Get a fixed value within range.
405 */
406 float getFixedValue(const float value) const noexcept
407 {
408 if (value <= min)
409 return min;
410 if (value >= max)
411 return max;
412 return value;
413 }
414
415 /**
416 Get a value normalized to 0.0<->1.0.
417 */
418 float getNormalizedValue(const float value) const noexcept
419 {
420 const float normValue = (value - min) / (max - min);
421
422 if (normValue <= 0.0f)
423 return 0.0f;
424 if (normValue >= 1.0f)
425 return 1.0f;
426 return normValue;
427 }
428
429 /**
430 Get a value normalized to 0.0<->1.0.
431 Overloaded function using double precision values.
432 */
433 double getNormalizedValue(const double& value) const noexcept
434 {
435 const double normValue = (value - min) / (max - min);
436
437 if (normValue <= 0.0)
438 return 0.0;
439 if (normValue >= 1.0)
440 return 1.0;
441 return normValue;
442 }
443
444 /**
445 Get a value normalized to 0.0<->1.0, fixed within range.
446 */
447 float getFixedAndNormalizedValue(const float value) const noexcept
448 {
449 if (value <= min)
450 return 0.0f;
451 if (value >= max)
452 return 1.0f;
453
454 const float normValue = (value - min) / (max - min);
455
456 if (normValue <= 0.0f)
457 return 0.0f;
458 if (normValue >= 1.0f)
459 return 1.0f;
460
461 return normValue;
462 }
463
464 /**
465 Get a value normalized to 0.0<->1.0, fixed within range.
466 Overloaded function using double precision values.
467 */
468 double getFixedAndNormalizedValue(const double value) const noexcept
469 {
470 if (value <= min)
471 return 0.0;
472 if (value >= max)
473 return 1.0;
474
475 const double normValue = (value - min) / (max - min);
476
477 if (normValue <= 0.0)
478 return 0.0;
479 if (normValue >= 1.0)
480 return 1.0;
481
482 return normValue;
483 }
484
485 /**
486 Get a proper value previously normalized to 0.0<->1.0.
487 */
488 float getUnnormalizedValue(const float value) const noexcept
489 {
490 if (value <= 0.0f)
491 return min;
492 if (value >= 1.0f)
493 return max;
494
495 return value * (max - min) + min;
496 }
497
498 /**
499 Get a proper value previously normalized to 0.0<->1.0.
500 Overloaded function using double precision values.
501 */
502 double getUnnormalizedValue(const double value) const noexcept
503 {
504 if (value <= 0.0)
505 return min;
506 if (value >= 1.0)
507 return max;
508
509 return value * (max - min) + min;
510 }
511};

544/**
545 Details around parameter enumeration values.@n
546 Wraps ParameterEnumerationValues and provides a few extra details to the host about these values.
547 */
548struct ParameterEnumerationValues {
549 /**
550 Number of elements allocated in @values.
551 */
552 uint8_t count;
553
554 /**
555 Whether the host is to be restricted to only use enumeration values.
556
557 @note This mode is only a hint! Not all hosts and plugin formats support this mode.
558 */
559 bool restrictedMode;
560
561 /**
562 Array of @ParameterEnumerationValue items.@n
563 When assigning this pointer manually, it must be allocated on the heap with `new ParameterEnumerationValue[count]`.@n
564 The array pointer will be automatically deleted later unless @p deleteLater is set to false.
565 */
566 ParameterEnumerationValue* values;
567
568 /**
569 Whether to take ownership of the @p values pointer.@n
570 Defaults to true unless stated otherwise.
571 */
572 bool deleteLater;
573
574 /**
575 Default constructor, for zero enumeration values.
576 */
577 constexpr ParameterEnumerationValues() noexcept
578 : count(0),
579 restrictedMode(false),
580 values(nullptr),
581 deleteLater(true) {}
582
583 /**
584 Constructor using custom values.@n
585 When using this constructor the pointer to @values MUST have been statically declared.@n
586 It will not be automatically deleted later.
587 */
588 constexpr ParameterEnumerationValues(uint8_t c, bool r, ParameterEnumerationValue* v) noexcept
589 : count(c),
590 restrictedMode(r),
591 values(v),
592 deleteLater(false) {}
593
594 // constexpr
595 ~ParameterEnumerationValues() noexcept
596 {
597 if (deleteLater)
598 delete[] values;
599 }
600
601 DAF_DECLARE_NON_COPYABLE(ParameterEnumerationValues)
602};

632 @note Parameter symbols MUST be unique within a plugin instance.

635

637 The unit of this parameter.@n
638 This means something like "dB", "kHz" and "ms".@n

642
643 /**
644 An extensive description/comment about the parameter.
645 @note This value is optional and only used for LV2.
646 */
647 String description;
648
649 /**
650 Ranges of this parameter.@n
651 The ranges describe the default, minimum and maximum values.
652 */

654
655 /**
656 Enumeration details.@n

658 */

660
661 /**
662 Designation for this parameter.
663 */

665
666 /**
667 MIDI CC to use by default on this parameter.@n

669 Must also be less or equal to 120.
670 @note This value is only a hint! Hosts might map it automatically or completely ignore it.

736 /**
737 Initialize a parameter for a specific designation.
738 */
739 void initDesignation(ParameterDesignation d) noexcept
740 {
741 designation = d;
742
743 switch (d)
744 {
745 case kParameterDesignationNull:
746 break;
747 case kParameterDesignationBypass:
748 hints = kParameterIsAutomatable|kParameterIsBoolean|kParameterIsInteger;
749 name = "Bypass";
750 shortName = "Bypass";
751 symbol = ParameterDesignationSymbols::bypass;
752 unit = "";
753 midiCC = 0;
754 groupId = kPortGroupNone;
755 ranges.def = 0.0f;
756 ranges.min = 0.0f;
757 ranges.max = 1.0f;
758 break;
759 case kParameterDesignationReset:
760 hints = kParameterIsAutomatable|kParameterIsBoolean|kParameterIsInteger|kParameterIsTrigger;
761 name = "Reset";
762 shortName = "Reset";
763 symbol = ParameterDesignationSymbols::reset;
764 unit = "";
765 midiCC = 0;
766 groupId = kPortGroupNone;
767 ranges.def = 0.0f;
768 ranges.min = 0.0f;
769 ranges.max = 1.0f;
770 break;
771 }
772 }

837 For example, two audio ports in a group may form a stereo stream.
838

844 */

849 The name doesn't have to be unique within a plugin instance, but it's recommended.
850 */
851 String name;
852
853 /**
854 The symbol of this port group.@n
855 A port group symbol is a short restricted name used as a machine and human readable identifier.@n
856 The first character must be one of _, a-z or A-Z and subsequent characters can be from _, a-z, A-Z and 0-9.
857 @note Port group symbols MUST be unique within a plugin instance.
858 */
859 String symbol;

872 /**
