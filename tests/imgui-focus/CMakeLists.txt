cmake_minimum_required(VERSION 3.15)
project(DafImGuiFocusTests LANGUAGES CXX)
enable_testing()
get_filename_component(DAF_ROOT "../.." ABSOLUTE)
set(DAF_LIBRARIES OFF CACHE BOOL "" FORCE)
set(DAF_EXAMPLES OFF CACHE BOOL "" FORCE)
set(DAF_WIDGETS_GALLERY OFF CACHE BOOL "" FORCE)
add_subdirectory("${DAF_ROOT}" daf)
daf__add_dgl_opengl3(TRUE TRUE TRUE)
daf__add_widgets_imgui(dgl-opengl3)
add_executable(imgui-focus "${DAF_ROOT}/tests/ImGuiFocus.cpp")
target_compile_features(imgui-focus PRIVATE cxx_std_11)
target_compile_definitions(imgui-focus PRIVATE DONT_SET_USING_DGL_NAMESPACE)
target_link_libraries(imgui-focus PRIVATE daf-widgets-imgui)
add_test(NAME imgui-focus COMMAND imgui-focus)
set_tests_properties(imgui-focus PROPERTIES TIMEOUT 30)
