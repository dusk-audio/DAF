cmake_minimum_required(VERSION 3.15)
project(DafNativeFileDragTests LANGUAGES CXX)
enable_testing()
get_filename_component(DAF_ROOT "../.." ABSOLUTE)
if(WIN32)
    add_executable(file-drag "${DAF_ROOT}/tests/FileDragWin.cpp")
    target_compile_features(file-drag PRIVATE cxx_std_11)
    target_compile_definitions(file-drag PRIVATE DONT_SET_USING_DGL_NAMESPACE)
    target_include_directories(file-drag PRIVATE "${DAF_ROOT}" "${DAF_ROOT}/tests"
        "${DAF_ROOT}/dgl/src/pugl-upstream/include")
    target_link_libraries(file-drag PRIVATE ole32 uuid user32 gdi32 comdlg32 dwmapi shell32)
    add_test(NAME native-file-payload COMMAND file-drag)
else()
    message(FATAL_ERROR "This standalone target runs the native Windows OLE payload test.")
endif()
