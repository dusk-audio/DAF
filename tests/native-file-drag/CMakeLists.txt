cmake_minimum_required(VERSION 3.15)
project(DafNativeFileDragTests LANGUAGES CXX)
enable_testing()
get_filename_component(DAF_ROOT "../.." ABSOLUTE)
if(WIN32)
    add_executable(file-drag "${DAF_ROOT}/tests/FileDragWin.cpp")
    target_compile_features(file-drag PRIVATE cxx_std_11)
    target_compile_definitions(file-drag PRIVATE DONT_SET_USING_DGL_NAMESPACE)
    target_include_directories(file-drag PRIVATE "${DAF_ROOT}" "${DAF_ROOT}/tests"
        "${DAF_ROOT}/dgl/src/pugl-upstream/include")
    target_link_libraries(file-drag PRIVATE ole32 uuid user32 gdi32 comdlg32 dwmapi shell32)
    add_test(NAME native-file-payload COMMAND file-drag)
    option(DAF_FILE_DRAG_INTERACTIVE "Test a real OLE drop on an unlocked interactive desktop" OFF)
    if(DAF_FILE_DRAG_INTERACTIVE)
        # Use the complete production DGL stub implementation, without renderer test shims.
        set(sdk_sources Application ApplicationPrivateData Color EventHandlers Geometry ImageBase
            ImageBaseWidgets Layout SubWidget SubWidgetPrivateData TopLevelWidget TopLevelWidgetPrivateData
            Widget WidgetPrivateData Window WindowPrivateData Stub pugl)
        foreach(source IN LISTS sdk_sources)
            list(APPEND sdk_files "${DAF_ROOT}/dgl/src/${source}.cpp")
        endforeach()
        add_executable(file-drag-native "${DAF_ROOT}/tests/FileDragWinNative.cpp" ${sdk_files})
        target_compile_features(file-drag-native PRIVATE cxx_std_11)
        target_compile_definitions(file-drag-native PRIVATE DONT_SET_USING_DGL_NAMESPACE NOMINMAX)
        target_include_directories(file-drag-native PRIVATE "${DAF_ROOT}" "${DAF_ROOT}/tests"
            "${DAF_ROOT}/dgl/src/pugl-upstream/include")
        target_link_libraries(file-drag-native PRIVATE ole32 uuid user32 gdi32 comdlg32 dwmapi shell32)
        add_test(NAME native-file-drop COMMAND file-drag-native)
        set_tests_properties(native-file-drop PROPERTIES TIMEOUT 20)
    endif()

elseif(APPLE)
    enable_language(OBJCXX)
    add_executable(file-drag "${DAF_ROOT}/tests/FileDragCocoa.cpp")
    set_source_files_properties("${DAF_ROOT}/tests/FileDragCocoa.cpp" PROPERTIES LANGUAGE OBJCXX)
    set_target_properties(file-drag PROPERTIES OBJCXX_STANDARD 11 OBJCXX_STANDARD_REQUIRED YES)
    target_compile_definitions(file-drag PRIVATE DONT_SET_USING_DGL_NAMESPACE)
    target_include_directories(file-drag PRIVATE "${DAF_ROOT}" "${DAF_ROOT}/tests"
        "${DAF_ROOT}/dgl/src/pugl-upstream/include")
    target_link_libraries(file-drag PRIVATE "-framework Cocoa" "-framework CoreVideo")
    add_test(NAME native-file-gesture COMMAND file-drag)
    option(DAF_FILE_DRAG_INTERACTIVE "Test a real Cocoa drop with existing event-posting access" OFF)
    if(DAF_FILE_DRAG_INTERACTIVE)
        add_executable(file-drag-native "${DAF_ROOT}/tests/FileDragCocoaNative.cpp")
        set_source_files_properties("${DAF_ROOT}/tests/FileDragCocoaNative.cpp" PROPERTIES LANGUAGE OBJCXX)
        set_target_properties(file-drag-native PROPERTIES OBJCXX_STANDARD 11 OBJCXX_STANDARD_REQUIRED YES)
        target_compile_definitions(file-drag-native PRIVATE DONT_SET_USING_DGL_NAMESPACE)
        target_include_directories(file-drag-native PRIVATE "${DAF_ROOT}" "${DAF_ROOT}/tests"
            "${DAF_ROOT}/dgl/src/pugl-upstream/include")
        target_link_libraries(file-drag-native PRIVATE "-framework Cocoa" "-framework CoreVideo" "-framework ApplicationServices")
        add_test(NAME native-file-drop COMMAND file-drag-native)
        set_tests_properties(native-file-drop PROPERTIES TIMEOUT 20)
        # DGL windows receiving DGL's own drag, the production stub implementation with pugl's Cocoa unity file
        set(sdk_sources Application ApplicationPrivateData Color EventHandlers Geometry ImageBase
            ImageBaseWidgets Layout SubWidget SubWidgetPrivateData TopLevelWidget TopLevelWidgetPrivateData
            Widget WidgetPrivateData Window WindowPrivateData Stub)
        foreach(source IN LISTS sdk_sources)
            list(APPEND sdk_files "${DAF_ROOT}/dgl/src/${source}.cpp")
        endforeach()
        add_executable(file-drop-native "${DAF_ROOT}/tests/FileDropCocoaNative.cpp" "${DAF_ROOT}/dgl/src/pugl.mm" ${sdk_files})
        set_source_files_properties("${DAF_ROOT}/tests/FileDropCocoaNative.cpp" "${DAF_ROOT}/dgl/src/pugl.mm"
            PROPERTIES LANGUAGE OBJCXX)
        set_target_properties(file-drop-native PROPERTIES CXX_STANDARD 11 OBJCXX_STANDARD 11 OBJCXX_STANDARD_REQUIRED YES)
        target_compile_definitions(file-drop-native PRIVATE DONT_SET_USING_DGL_NAMESPACE)
        target_include_directories(file-drop-native PRIVATE "${DAF_ROOT}" "${DAF_ROOT}/tests"
            "${DAF_ROOT}/dgl/src/pugl-upstream/include")
        target_link_libraries(file-drop-native PRIVATE "-framework Cocoa" "-framework CoreVideo" "-framework ApplicationServices")
        add_test(NAME native-file-drop-target COMMAND file-drop-native)
        set_tests_properties(native-file-drop-target PROPERTIES TIMEOUT 30)
    endif()
else()
    message(FATAL_ERROR "These standalone tests require native Windows or macOS.")
endif()
