33/**
34 This set of static variables act as a build sentinel that detects a configuration error.
35
36 Usually this means the way DGL was built and how it is being used and linked into your program is different,
37 we want to avoid such combinations as memory layout would then also be different
38 leading to all sort of subtle but very nasty memory corruption issues.
39
40 Make sure the flags used to build DGL match the ones used by your program and the link errors should go away.
41 */

43 static struct DAF_JOIN_MACRO(_, NAME) { bool ok; DAF_JOIN_MACRO(_, NAME)() noexcept; } NAME;

79/**
80 Base DGL Application class.
81
82 One application instance is required for creating a window.
83 There's no single/global application instance in DGL, and multiple windows can share the same app instance.
84
85 In standalone mode an application will automatically quit its event-loop when all its windows are closed.
86
87 Unless stated otherwise, functions within this class are not thread-safe.
88 */
89class DAF_API Application
90{
91public:
92 /**
93 Type of application to setup, either "classic" or "modern".
94
95 What this means depends on the OS.
96 */
97 enum Type {
98 kTypeAuto,
99 kTypeClassic,
100 kTypeModern,
101 };
102
103 /**
104 Windowing system to open the application's windows on.
105
106 Only Linux and the other X11/Wayland systems have one to choose. A DGL_BACKEND=both build picks X11 or
107 Wayland at runtime, by the rule in dgl/src/runtime-backend.md, unless the application asks for one.
108 The request is final: if that system cannot be used, or the build does not have it, no window opens
109 and a message says why, rather than the application ending up on the other one.
110 It takes the place of the DAF_WINDOW_BACKEND environment variable.
111
112 On macOS, Windows and the other platforms with a single windowing system the request is ignored.
113 */
114 enum WindowSystem {
115 kWindowSystemAuto,
116 kWindowSystemX11,
117 kWindowSystemWayland,
118 };
119
120 /**
121 Constructor for standalone or plugin application.
122 */
123 Application(bool isStandalone = true, Type type = kTypeAuto, WindowSystem windowSystem = kWindowSystemAuto);
124
125 /**
126 Constructor for a standalone application.
127 This specific constructor is required if using web views in standalone applications.
128 On Linux a web view needs X11, see WindowSystem.
129 */
130 Application(int argc, char* argv[], WindowSystem windowSystem = kWindowSystemAuto);
131
132 /**
133 Destructor.
134 */
135 virtual ~Application();
136
137 /**
138 Idle function.
139 This runs the application event-loop once.
140 */
141 void idle();
142
143 /**
144 Run the application event-loop until all Windows are closed.
145 idle() is called at regular intervals.
146 @note This function is meant for standalones only, *never* call this from plugins.
147 */
148 void exec(uint idleTimeInMs = 30);
149
150 /**
151 Quit the application.
152 This stops the event-loop and closes all Windows.
153 This function is thread-safe.
154 */
155 void quit();
156
157 /**
158 Check if the application is about to quit.
159 Returning true means there's no event-loop running at the moment (or it's just about to stop).
160 This function is thread-safe.
161 */
162 bool isQuitting() const noexcept;
163
164 /**
165 Check if the application is standalone, otherwise running as a module or plugin.
166 This function is thread-safe.
167 */
168 bool isStandalone() const noexcept;
169
170 /**
171 Return the time in seconds.
172
173 This is a monotonically increasing clock with high resolution.@n
174 The returned time is only useful to compare against other times returned by this function,
175 its absolute value has no meaning.
176 */
177 double getTime() const;
178
179 /**
180 Return the application type, either kTypeClassic or kTypeModern.
181 This function never return kTypeAuto.
182 */
183 Type getType() const noexcept;
184
185 /**
186 Return the windowing system the application's windows open on, kWindowSystemX11 or kWindowSystemWayland.
187 Returns kWindowSystemAuto where the platform has no choice of one, and when none could be opened.
188 */
189 WindowSystem getWindowSystem() const noexcept;
190
191 /**
192 Add a callback function to be triggered on every idle cycle.
193 You can add more than one, and remove them at anytime with removeIdleCallback().
194 Idle callbacks trigger right after OS event handling and Window idle events (within the same cycle).
195 There are no guarantees in terms of timing, use Window::addIdleCallback for time-relative callbacks.
196 Registrations added during dispatch start on the next idle call. Removing a registration
197 cancels its pending call immediately; nested idle calls do not re-enter an active registration.
198 */
199 void addIdleCallback(IdleCallback* callback);
200
201 /**
202 Remove an idle callback previously added via addIdleCallback().
203 */
204 void removeIdleCallback(IdleCallback* callback);
205
206 /**
207 Get the class name of the application.
208
209 This is a stable identifier for the application, used as the window class/instance name on X11 and Windows.
210 It is not displayed to the user, but can be used in scripts and by window managers,
211 so it should be the same for every instance of the application, but different from other applications.
212
213 Plugins created with DAF have their class name automatically set based on DGL_NAMESPACE and plugin name.
214 */
215 const char* getClassName() const noexcept;
216
217 /**
218 Set the class name of the application.
219 @see getClassName
220 */
221 void setClassName(const char* name);
222
223private:
224 struct PrivateData;
225 PrivateData* const pData;
226 friend class Window;
227 #ifdef DAF_NAMESPACE
228 friend class DAF_NAMESPACE::PluginApplication;
229 #endif
230
231 DAF_DECLARE_NON_COPYABLE_WITH_LEAK_DETECTOR(Application)
232};
