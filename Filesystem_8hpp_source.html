35/*
36 * Wrapper around `fopen` call, needed on Windows because its C standard functions use ASCII instead of UTF-8.
37 */
38static inline
39FILE* d_fopen(const char* const pathname, const char* const mode)
40{
41 #ifdef DAF_OS_WINDOWS
42 WCHAR lpathname[MAX_PATH];
43 WCHAR lmode[4];
44 if (MultiByteToWideChar(CP_UTF8, 0, pathname, -1, lpathname, ARRAY_SIZE(lpathname)) != 0 &&
45 MultiByteToWideChar(CP_UTF8, 0, mode, -1, lmode, ARRAY_SIZE(lmode)) != 0)
46 return _wfopen(lpathname, lmode);
47 #endif
48
49 return fopen(pathname, mode);
50}

55/**
56 Handy class to help write files in a safe way, which does:
57 - open pathname + ".tmp" instead of opening a file directly (so partial writes are safe)
58 - on close, flush data to disk and rename file to remove ".tmp"
59
60 To use it, create a local variable (on the stack) and call ok() or manually check @a fd variable.
61 @code
62 if (const SafeFileWriter file("/path/to/file.txt"); file.ok())
63 file.write("Success!");
64 @endcode
65 */
66struct SafeFileWriter
67{
68 String filename;
69 FILE* const fd;
70
71 /**
72 Constructor, opening @a pathname + ".tmp" for writing.
73 */
74 SafeFileWriter(const char* const pathname, const char* const mode = "w")
75 : filename(pathname),
76 fd(d_fopen(filename + ".tmp", mode))
77 {
78 #ifndef DAF_OS_WINDOWS
79 if (fd == nullptr)
80 d_stderr2("failed to open '%s' for writing: %s", pathname, std::strerror(errno));
81 #endif
82 }
83
84 /**
85 Destructor, will flush file data contents, close and rename file.
86 */
87 ~SafeFileWriter()
88 {
89 if (fd == nullptr)
90 return;
91
92 std::fflush(fd);
93 std::fclose(fd);
94 std::rename(filename + ".tmp", filename);
95 }
96
97 /** Check if the file was opened successfully. */
98 inline bool ok() const noexcept
99 {
100 return fd != nullptr;
101 }
102
103 /** Wrapper around `fwrite`, purely for convenience. */
104 inline size_t write(const void* const ptr, const size_t size, const size_t nmemb = 1) const
105 {
106 return std::fwrite(ptr, size, nmemb, fd);
107 }
108};
