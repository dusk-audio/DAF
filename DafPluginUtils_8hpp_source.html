109#if DAF_PLUGIN_NUM_OUTPUTS > 0
110/**
111 Handy class to help keep audio buffer in sync with incoming MIDI events.
112 To use it, create a local variable (on the stack) and call nextEvent() until it returns false.
113 @code
114 for (AudioMidiSyncHelper amsh(outputs, frames, midiEvents, midiEventCount); amsh.nextEvent();)
115 {
116 float* const outL = amsh.outputs[0];
117 float* const outR = amsh.outputs[1];
118
119 for (uint32_t i=0; i<amsh.midiEventCount; ++i)
120 {
121 const MidiEvent& ev(amsh.midiEvents[i]);
122 // ... do something with the midi event
123 }
124
125 renderSynth(outL, outR, amsh.frames);
126 }
127 @endcode
128
129 Some important notes when using this class:
130 1. MidiEvent::frame retains its original value, but it is useless, do not use it.
131 2. The class variable names are the same as the default ones in the run function.
132 Keep that in mind and try to avoid typos. :)
133 */
134struct AudioMidiSyncHelper
135{
136 /** Parameters from the run function, adjusted for event sync */
137 float* outputs[DAF_PLUGIN_NUM_OUTPUTS];
138 uint32_t frames;
139 const MidiEvent* midiEvents;
140 uint32_t midiEventCount;
141
142 /**
143 Constructor, using values from the run function.
144 */
145 AudioMidiSyncHelper(float** const o, uint32_t f, const MidiEvent* m, uint32_t mc)
146 : outputs(),
147 frames(0),
148 midiEvents(m),
149 midiEventCount(0),
150 remainingFrames(f),
151 remainingMidiEventCount(mc),
152 totalFramesUsed(0)
153 {
154 for (uint i=0; i<DAF_PLUGIN_NUM_OUTPUTS; ++i)
155 outputs[i] = o[i];
156 }
157
158 /**
159 Process a batch of events until no more are available.
160 You must not read any more values from this class after this function returns false.
161 */
162 bool nextEvent()
163 {
164 // nothing else to do
165 if (remainingFrames == 0)
166 return false;
167
168 // initial setup, need to find first MIDI event
169 if (totalFramesUsed == 0)
170 {
171 // no MIDI events at all in this process cycle
172 if (remainingMidiEventCount == 0)
173 {
174 frames = remainingFrames;
175 remainingFrames = 0;
176 totalFramesUsed += frames;
177 return true;
178 }
179
180 // render audio until first midi event, if needed
181 if (const uint32_t firstEventFrame = midiEvents[0].frame)
182 {
183 DAF_SAFE_ASSERT_UINT2_RETURN(firstEventFrame < remainingFrames,
184 firstEventFrame, remainingFrames, false);
185 frames = firstEventFrame;
186 remainingFrames -= firstEventFrame;
187 totalFramesUsed += firstEventFrame;
188 return true;
189 }
190 }
191 else
192 {
193 for (uint32_t i=0; i<DAF_PLUGIN_NUM_OUTPUTS; ++i)
194 if (outputs[i] != nullptr) // an EXTRA_IO layout narrower than the declared outputs leaves the rest null
195 outputs[i] += frames;
196 }
197
198 // no more MIDI events available
199 if (remainingMidiEventCount == 0)
200 {
201 frames = remainingFrames;
202 midiEvents = nullptr;
203 midiEventCount = 0;
204 remainingFrames = 0;
205 totalFramesUsed += frames;
206 return true;
207 }
208
209 // if there were midi events before, increment pointer
210 if (midiEventCount != 0)
211 midiEvents += midiEventCount;
212
213 const uint32_t firstEventFrame = midiEvents[0].frame;
214 DAF_SAFE_ASSERT_UINT2_RETURN(firstEventFrame >= totalFramesUsed,
215 firstEventFrame, totalFramesUsed, false);
216
217 midiEventCount = 1;
218 while (midiEventCount < remainingMidiEventCount)
219 {
220 if (midiEvents[midiEventCount].frame == firstEventFrame)
221 ++midiEventCount;
222 else
223 break;
224 }
225
226 frames = firstEventFrame - totalFramesUsed;
227 remainingFrames -= frames;
228 remainingMidiEventCount -= midiEventCount;
229 totalFramesUsed += frames;
230 return true;
231 }
232
233private:
234 /** @internal */
235 uint32_t remainingFrames;
236 uint32_t remainingMidiEventCount;
237 uint32_t totalFramesUsed;
238};
239#endif
