34/**
35 Web view options, for customizing web view details.
36*/
37struct WebViewOptions {
38 /**
39 Position offset, for cases of mixing regular widgets with web views.
40 */
41 struct PositionOffset {
42 /** Horizontal offset, with scale factor pre-applied */
43 int x;
44
45 /** Vertical offset, with scale factor pre-applied */
46 int y;
47
48 /** Constructor for default values */
49 PositionOffset() : x(0), y(0) {}
50 } offset;
51
52 /**
53 Set some JavaScript to evaluate on every new page load.
54 */
55 const char* initialJS;
56
57 /**
58 Message callback triggered from JavaScript code inside the WebView.
59 */
60 WebViewMessageCallback callback;
61 void* callbackPtr;
62
63 /** Constructor for default values */
64 WebViewOptions()
65 : offset(),
66 initialJS(nullptr),
67 callback(nullptr),
68 callbackPtr(nullptr) {}
69
70 /** Constructor providing a callback */
71 WebViewOptions(const WebViewMessageCallback cb, void* const ptr)
72 : offset(),
73 initialJS(nullptr),
74 callback(cb),
75 callbackPtr(ptr) {}
76};

80/**
81 Create a new web view.
82
83 The web view will be added on top of an existing platform-specific view/window.
84 This means it will draw on top of whatever is below it,
85 something to take into consideration if mixing regular widgets with web views.
86
87 Provided metrics must have scale factor pre-applied.
88
89 @p url: The URL to open, assumed to be in encoded form (e.g spaces converted to %20)
90 @p windowId: The native window id to attach this view to (X11 Window, HWND or NSView*)
91 @p scaleFactor: Scale factor in use
92 @p options: Extra options, optional
93*/
94WebViewHandle webViewCreate(const char* url,
95 uintptr_t windowId,
96 uint initialWidth,
97 uint initialHeight,
98 double scaleFactor,
99 const WebViewOptions& options = WebViewOptions());
100
101/**
102 Destroy the web view, handle must not be used afterwards.
103*/
104void webViewDestroy(WebViewHandle webview);
105
106/**
107 Idle the web view, to be called on regular intervals.
108 Can cause callbacks to trigger.
109*/
110void webViewIdle(WebViewHandle webview);
111
112/**
113 Evaluate/run JavaScript on the web view.
114*/
115void webViewEvaluateJS(WebViewHandle webview, const char* js);
116
117/**
118 Reload the web view current page.
119*/
120void webViewReload(WebViewHandle webview);

122/**
123 Resize the web view.

125void webViewResize(WebViewHandle webview, uint width, uint height, double scaleFactor);
